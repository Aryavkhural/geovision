#include <iostream>
using namespace std;
class BankAccount {
    private:
    string accountHolderName;

int accountNumber;

double balance;
public:
 BankAccount (){
     accountHolderName = "Unknown";
     accountNumber= 0;
     balance = 0;
 }
 BankAccount ( string name, int accNo,double initialBalance ){
    accountHolderName=name;
    accountNumber=accNo;
    balance=initialBalance;
 }
void deposit(double amount)
{
    if(amount>0){
        balance = amount+balance;
    }else{
        cout <<"error"<<endl;
    }
    cout<<"your balance is "<<balance;
}
void display(){
    cout <<"ACCOUNTHOLDERNAME:"<<accountHolderName<<endl;
    cout <<"ACCOUNTNO.:" <<accountNumber<<endl;
    cout <<"BALANCE:" <<balance<<endl;
    
}
};
